cd ~/ethlab

# ---------- network ID, chain ID, genesis hash (must be IDENTICAL on all 3) ----------
for n in 1 2 3; do
  echo "===== node$n ====="
  echo -n "net.version : "; geth attach --exec "net.version" node$n/geth.ipc
  echo -n "chainId     : "; geth attach --exec "admin.nodeInfo.protocols.eth.config.chainId" node$n/geth.ipc
  echo -n "genesis hash: "; geth attach --exec "eth.getBlock(0).hash" node$n/geth.ipc
  echo -n "peerCount   : "; geth attach --exec "net.peerCount" node$n/geth.ipc
  echo -n "blockNumber : "; geth attach --exec "eth.blockNumber" node$n/geth.ipc
  echo -n "latest hash : "; geth attach --exec "eth.getBlock('latest').hash" node$n/geth.ipc
done
